cmake_minimum_required(VERSION 3.20)
project(forge_nr_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(dlss5nr_bridge SHARED nr/bridge.cpp upstream/native/nvof_flow.cpp)
target_include_directories(dlss5nr_bridge PRIVATE upstream/native)
target_link_libraries(dlss5nr_bridge PRIVATE d3d12 d3d11 dxgi ole32)
add_library(nr_caller SHARED nr/caller.cpp)
target_include_directories(nr_caller PRIVATE upstream/native)
set_target_properties(nr_caller PROPERTIES OUTPUT_NAME "nvngx.dll_comfy")
if(MSVC)
  target_compile_options(dlss5nr_bridge PRIVATE /O2)
  target_compile_options(nr_caller PRIVATE /Od)
endif()
